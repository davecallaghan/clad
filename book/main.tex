% Formal statements carried in from the CLAD corpus keep their own designations
% --- Theorem 3b, Definition 3.2, Axiom 5 --- because the text cross-references
% them by number in nearly two hundred places. amsthm's automatic numbering
% would renumber them all, so these environments are unnumbered and take the
% original designation as the optional argument, which the style prints where a
% number would go. Same head and body treatment as the environments above, so
% the two halves of the book look alike.
\newtheoremstyle{cladplain}{\topsep}{\topsep}{\itshape}{}%
  {\bfseries}{.}{ }{\thmname{#1}\thmnote{ #3}}
\newtheoremstyle{claddefn}{\topsep}{\topsep}{\normalfont}{}%
  {\bfseries}{.}{ }{\thmname{#1}\thmnote{ #3}}
\theoremstyle{cladplain}
\newtheorem*{cthm}{Theorem}
\newtheorem*{clem}{Lemma}
\newtheorem*{ccor}{Corollary}
\newtheorem*{cprop}{Proposition}

\newtheorem*{cdef}{Definition}
\newtheorem*{cprp}{Property}
\newtheorem*{creq}{Requirement}
\newtheorem*{cprn}{Principle}
\newtheorem*{cinv}{Invariant}
